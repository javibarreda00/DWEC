Ejercicio 3

Coerción y comparaciones

function ejercicio3() {
  console.log("--- Ejercicio 3 · Coerción y comparaciones ---");

  
  console.log('"5" - 2 →', "5" - 2);   // El texto se convierte en número
  console.log('"5" + 2 →', "5" + 2);   // El número se junta con el texto
  console.log("true + 1 →", true + 1);   // true equivale a 1
  console.log('"1" + 3 + 5 →', "1" + 3 + 5);   // Se van juntando como texto
  console.log('2 + 6 + "5" →', 2 + 6 + "5");   // Primero suma y después junta
  console.log('"chao" - 1 →', "chao" - 1);   // No puede restar y devuelve NaN
  
  
  console.log('5 == "5" →', 5 == "5");     // Devuelve true. el == solo compara el valor


  console.log('5 === "5" →', 5 === "5");   // Devuelve false. el === compara tanto el valor como el tipo 

  
  console.log("0 == false →", 0 == false); // son iguales porque solo comparan el valor

  
  console.log("0 === false →", 0 === false); // son distintos porque sus tipos son diferentes

  
  console.log("null == undefined →", null == undefined); //son iguales porque solo comparan valor

  
  console.log("null === undefined →", null === undefined); // son distintos porque sus tipos son diferentes
}
